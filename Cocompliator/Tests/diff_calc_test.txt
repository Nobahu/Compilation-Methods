# Тест сложных формул
int a = 5;
int b = 3;
int c = 10;
int d = 2;
int result;

result = (a + b) * (c - d) / (a - 1);
write("Результат: ");
write(result);

# Проверка унарного минуса и вложенных скобок
int neg;
neg = -a + (b - (c + d)) * 2;
write("Унарный минус и вложенные скобки: ");
write(neg);

# Проверка арифметики с большим выражением
int big;
big = (a * b - c) / (d + 1) + (a + b * c) - 20;
write("Большое выражение: ");
write(big);
