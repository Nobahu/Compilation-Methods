int n = 3;
int i = 0;
int sum = 0;

int[n] arr;

while (i < n) {
    arr[i] = i * 2;
    i++; 
}

int index = 0;
while (index < n) {
    sum = sum + arr[index];
    index++;
}
write(sum);

root = sqrt(sum + 3);
write(root);

zero = 0;
valExp = exp(zero);
write(valExp);

angle = 0;

valSin = sin(angle);
write(valSin); 

valCos = cos(angle);
write(valCos);
